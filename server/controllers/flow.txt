How it works Initial authentication: A client authenticates with a server, often using credentials or an identity provider.Token issuance: Upon successful authentication, the server issues a bearer token to the client.Subsequent requests: The client includes the token in the Authorization header of every subsequent request to a protected resource, formatted as Authorization: Bearer <token>.Server validation: The server receives the request, validates the token (checking for authenticity and expiration), and if valid, grants access to the resource. If the token is invalid, it will typically return a \(401\) Unauthorized response.

exports.getTopCourses= async (req, res) => {
  try {
    const limit = Number(req.query.limit) || 4;

    const data = await Course.aggregate([
      { $match: { status: "PUBLISHED" } },

      {
        $project: {
          category: 1,
          year: { $year: "$createdAt" },
          month: { $month: "$createdAt" },
          enrollments: {
            $size: { $ifNull: ["$studentEnrolled", []] },
          },
        },
      },

      {
        $group: {
          _id: {
            category: "$category",
            year: "$year",
            month: "$month",
          },
          totalEnrollments: { $sum: "$enrollments" },
        },
      },

      { $sort: { totalEnrollments: -1 } },
      { $limit: limit },

      {
        $lookup: {
          from: "categories",
          localField: "_id",
          foreignField: "_id",
          as: "CategoryData",
        },
      },

      { $unwind: "$CategoryData" },

      {
        $project: {
          _id: 0,
          categoryId: "$CategoryData._id",
          categoryName: "$CategoryData.name",
          year: "$_id.year",
          month: "$_id.month",
          totalEnrollments: 1,
        },
      },
    ]);

    res.json({ success: true, data });
  } catch (err) {
    res.status(500).json({ success: false, message: "Failed To Fetch" });
  }
};
