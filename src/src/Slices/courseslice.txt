import { createSlice } from "@reduxjs/toolkit";

const savedStep = localStorage.getItem("courseStep");

const initialState = {
  step: savedStep ? Number(savedStep) : 1,
  courseId: null,
  sections: [],
};

const courseSlice = createSlice({
  name: "course",
  initialState,
  reducers: {
    setStep: (state, action) => {
      state.step = action.payload;
      localStorage.setItem("courseStep", action.payload); //  persist step only
    },
    setCourseId: (state, action) => {
      state.courseId = action.payload;
    },
    setSections: (state, action) => {
      state.sections = action.payload;
    },
    resetCourse: (state) => {
      state.courseId = null;
      state.sections = [];
      state.step = 1;
      localStorage.removeItem("courseStep"); //  reset step
    },
  },
});

export const {
  setStep,
  setCourseId,
  setSections,
  resetCourse,
} = courseSlice.actions;

export default courseSlice.reducer;
