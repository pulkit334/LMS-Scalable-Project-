import { useDispatch, useSelector } from "react-redux";
import CourseInformation from "../components/core/coursePages/course1";
import CourseBuilder from "../components/core/coursePages/CourseBuilder";
import PublishCourse from "../components/core/coursePages/PublishCourse";
import {
  setStep,
  setCourseId,
  setSections,
} from "../Slices/courseSlice";
  
const Course = () => {
  const dispatch = useDispatch();

  const { step, courseId, sections } = useSelector(
    (state) => state.course
  );


  return (
    
    <>
    
      {step === 1 && (
        <CourseInformation
          onSuccess={(id) => {
            dispatch(setCourseId(id));
            dispatch(setStep(2));
          }}
        />
      )}

      {step === 2 && (
        <CourseBuilder
          courseId={courseId}
          sections={sections}
          setSections={(data) => dispatch(setSections(data))}
          onBack={() => dispatch(setStep(1))}
          onNext={() => dispatch(setStep(3))}
        />
      )}

      {step === 3 && (
        <PublishCourse
          courseId={courseId}
          onBack={() => dispatch(setStep(2))}
        />
      )}
    </>
  );
};

export default Course;
